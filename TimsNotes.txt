# Link my vimfiles from My.Configs to my profile.
  New-Item -Type SymbolicLink -Name vimfiles -Path C:\Users\MussonT\ -Value C:\usr\My.Configs\vimfiles\

# Link my vimfiles from My.Configs to my nvim location.
  New-Item -Type SymbolicLink -Name nvim -Path C:\Users\MussonT\AppData\Local\ -Value C:\usr\My.Configs\vimfiles\

# Link my vimrc (for gvim) to init.vim (for neovim)
  New-Item -Type SymbolicLink -Name init.vim -Path C:\usr\My.Configs\vimfiles\ -Value C:\usr\My.Configs\vimfiles\vimrc
